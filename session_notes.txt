<?php

// =============================================================
//  Helper: Get real client IP (your header order)
// =============================================================
function get_client_ip() {
    $headers = array(
        'HTTP_CF_CONNECTING_IP',
        'HTTP_TRUE_CLIENT_IP',
        'HTTP_CLIENT_IP',
        'HTTP_X_FORWARDED_FOR',
        'HTTP_X_FORWARDED',
        'HTTP_X_CLUSTER_CLIENT_IP',
        'HTTP_FORWARDED_FOR',
        'HTTP_FORWARDED',
        'HTTP_X_REAL_IP',
        'REMOTE_ADDR'
    );

    foreach ($headers as $h) {
        if (!empty($_SERVER[$h])) {
            $ip_list = $_SERVER[$h];
            if (strpos($ip_list, ',') !== false) {
                $ips = array_map('trim', explode(',', $ip_list));
                $ip = $ips[0];
            } else {
                $ip = $ip_list;
            }

            if (filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RESERVED_RANGE)) {
                return $ip;
            }
        }
    }
    return $_SERVER['REMOTE_ADDR'] ?? '0.0.0.0';
}

// =============================================================
//  Helper: Class C IP
// =============================================================
function get_class_c_ip($ip = null) {
    if ($ip === null) $ip = get_client_ip();

    if (filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_IPV6)) {
        $parts = explode(':', $ip);
        return implode(':', array_slice($parts, 0, 4)) . '::';
    }

    return preg_replace('/\.\d{1,3}$/', '.0', $ip);
}

// =============================================================
//  Helper: YOUR exact UA normalization (a-zA-Z only, max 100 chars)
// =============================================================
function normalize_ua() {
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? '';
    
    // Remove everything that is NOT a-z or A-Z
    $ua = preg_replace('/[^a-zA-Z]/', '', $ua);
    
    // Your 100 char limit
    return substr($ua, 0, 100);
}

// =============================================================
//  CREATE SESSION
// =============================================================
function create_session($class_c, $user_id) {
    $ua         = normalize_ua();
    $class_c_ip = get_class_c_ip();
    $random     = bin2hex(random_bytes(32));

    $input = $class_c . '|' . $user_id . '|' . $ua . '|' . $class_c_ip . '|' . $random;
    $session_id = hash('sha256', $input);

    $data = [
        'session_id'    => $session_id,
        'user_id'       => $user_id,
        'class_c'       => $class_c,
        'user_agent'    => $ua,                    // now exactly your 100-char version
        'class_c_ip'    => $class_c_ip,
        'random_token'  => $random,
        'created'       => time(),
        'last_activity' => time(),
        'expires'       => time() + (86400 * 14)
    ];

    db_insert_session($data);

    setcookie('sess', $session_id, [
        'expires'  => time() + (86400 * 14),
        'path'     => '/',
        'secure'   => true,
        'httponly' => true,
        'samesite' => 'Strict'
    ]);

    return $session_id;
}

// =============================================================
//  VALIDATE SESSION
// =============================================================
function validate_session() {
    $cookie_id = $_COOKIE['sess'] ?? '';
    if (empty($cookie_id) || strlen($cookie_id) !== 64) {
        return false;
    }

    $row = db_get_session($cookie_id);
    if (!$row) return false;

    $current_ua         = normalize_ua();
    $current_class_c_ip = get_class_c_ip();

    if ($row['user_id']    != $expected_user_id || 
        $row['class_c']    != $expected_class_c ||
        !hash_equals($row['user_agent'], $current_ua) ||
        !hash_equals($row['class_c_ip'], $current_class_c_ip)) {

        destroy_session($cookie_id);
        return false;
    }

    if (time() - $row['last_activity'] > 1800) {   // 30 min idle
        destroy_session($cookie_id);
        return false;
    }

    if (should_regenerate_session($row)) {
        regenerate_session($row);
    }

    db_update_session_activity($cookie_id, time());

    return $row;
}
